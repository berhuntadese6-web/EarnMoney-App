<script>
let currentUserEmail = "";
let userPoints = 0;

function loginUser() {
const email = document.getElementById('emailInput').value.trim();
if (!email || !email.includes('@')) {
alert('እባክዎ ትክክለኛ ኢሜይል ያስገቡ!');
return;
}
currentUserEmail = email;
document.getElementById('displayEmail').innerText = email;

// የኢሜይል ሳጥኑን ደብቆ የስራውን ገጽ ማሳየት
document.getElementById('authSection').style.display = 'none';
document.getElementById('mainSection').style.display = 'block';
}

async function verifyTask() {
const proof = document.getElementById('proofText').value;
const resultBox = document.getElementById('resultBox');

if (!proof) {
alert('እባክዎ የማስረጃ ጽሑፍ ያስገቡ!');
return;
}

resultBox.style.display = 'block';
resultBox.innerHTML = 'AI ማስረጃውን እያረጋገጠ ነው... እባክዎ ይጠብቁ።';

try {
const response = await fetch('http://localhost:8000/api/ai/verify-proof', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ user_id: currentUserEmail, proof_text: proof })
});

const data = await response.json();

if (data.status === "approved") {
userPoints += data.reward_points;
document.getElementById('userPoints').innerText = userPoints;
resultBox.innerHTML = ✅ **ተሳክቷል!**\nመልዕክት: ${data.message}\nየተሸለሙት ነጥብ: +${data.reward_points};
} else {
resultBox.innerHTML = ❌ **አልተሳካም**\nመልዕክት: ${data.message};
}
} catch (error) {
resultBox.innerHTML = ⚠️ ስህተት አጋጥሟል: ሰርቨሩ ላይ መገናኘት አልተቻለም።;
console.error(error);
}
}
</script>
